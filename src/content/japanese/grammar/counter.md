  - attaches_to: "people"
    form: "〜人{にん}"
    example: "一人{ひとり}、二人{ふたり}、三人{さんにん}"
  - attaches_to: "flat things"
    form: "〜枚{まい}"
    example: "切符{きっぷ}を二枚{にまい}"
  - attaches_to: "long things"
    form: "〜本{ほん}"
    example: "ペンを三本{さんぼん}"
  - attaches_to: "times / occurrences"
    form: "〜回{かい}"
    example: "二回{にかい}"
  - attaches_to: "small animals"
    form: "〜匹{ひき}"
    example: "猫{ねこ}を一匹{いっぴき}"
usageSetting: "The number + counter phrase usually sits just before the verb (or after the noun + が/を); pick the counter by the type or shape of what you count."

notes:
