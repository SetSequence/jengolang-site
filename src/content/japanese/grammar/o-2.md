contrasts:
  - slug: "de"
    label: "で"
    distinction: "を marks the route moved along (公園{こうえん}を歩{ある}く = 'walk through the park'); で marks the place an action happens (公園{こうえん}で遊{あそ}ぶ = 'play in the park'). を = passing through, で = located at."

    en: "Go straight along this road."

    en: "I run along the river every morning."
notes:
