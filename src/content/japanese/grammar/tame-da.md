senses:
  - label: "it is because"
    equivalents:
      - "it is because of ~ / the reason is ~ — closes a sentence by naming the cause"
    keySentence:
      jp: "電車{でんしゃ}が遅{おく}れたのは事故{じこ}のためだ。"
      en: "The train was late because of an accident."
    examples:
      - jp: "店{みせ}が閉{し}まっているのは祝日{しゅくじつ}のためだ。"
        en: "The shop is closed because it's a holiday."
  - label: "it is for the sake of"
    equivalents:
      - "it is for the sake of ~ / it's for ~ — closes a sentence by naming the purpose or beneficiary"
    keySentence:
      jp: "厳{きび}しく言{い}うのは君{きみ}のためだ。"
      en: "I'm being strict for your own sake."
    examples:
      - jp: "貯金{ちょきん}しているのは家{いえ}を買{か}うためだ。"
        en: "I'm saving up in order to buy a house."
formation:
  - attaches_to: "Noun (+の) / Verb (plain)"
    form: "N + のためだ; V(plain) + ためだ"
    example: "病気{びょうき}のためだ / 合格{ごうかく}するためだ"
contrasts:
  - slug: "kara-da"
    label: "からだ"
    distinction: "Both can close a sentence with 'it's because ~.' ためだ is a touch more formal and written, and uniquely also carries the 'for the sake of' purpose reading, which からだ cannot."
  - slug: "tame-ni"
    label: "〜ために"
    distinction: "ために links a reason/purpose into a following clause mid-sentence (合格{ごうかく}するために勉強{べんきょう}する). ためだ is the sentence-final predicate form — 'it is because / it is for ~.'"
  - slug: "wake-da"
    label: "〜わけだ"
    distinction: "ためだ states the cause directly ('it's because of X'). わけだ draws a conclusion that logically follows ('that's why ~ / so it makes sense that ~') — a deduction, not the raw reason."
