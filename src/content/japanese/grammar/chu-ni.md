senses:
  - label: "in-the-middle"
    equivalents:
      - "in the middle of ~ / during ~ — while an action or period is in progress"
    keySentence:
      jp: "会議中{かいぎちゅう}は携帯{けいたい}を切{き}ってください。"
      en: "Please turn off your phone during the meeting."
    examples:
      - jp: "食事中{しょくじちゅう}に電話{でんわ}が鳴{な}った。"
        en: "The phone rang in the middle of the meal."
      - jp: "授業中{じゅぎょうちゅう}に居眠{いねむ}りしてしまった。"
        en: "I dozed off during class."
  - label: "within-a-period"
    equivalents:
      - "within ~ / by the end of ~ (sometime inside the period) — a deadline expressed as a span"
    keySentence:
      jp: "今日中{きょうじゅう}に返事{へんじ}をします。"
      en: "I'll give you an answer by the end of today."
    examples:
      - jp: "今週中{こんしゅうちゅう}にレポートを出{だ}さなければならない。"
        en: "I have to hand in the report by the end of this week."
      - jp: "午前中{ごぜんちゅう}に着{つ}く予定{よてい}です。"
        en: "I'm scheduled to arrive sometime in the morning."
formation:
  - attaches_to: "Noun (action or period)"
    form: "N + 中{ちゅう}(に)"
    example: "仕事中{しごとちゅう} / 工事中{こうじちゅう} / 今月中{こんげつちゅう}に"
contrasts:
  - slug: "aida-ni"
    label: "〜間に"
    distinction: "間{あいだ}に attaches to a clause or noun to mark a point within a span (留守{るす}の間{あいだ}に). 〜中{ちゅう} attaches directly to a noun and labels the noun itself as 'ongoing' (留守中{るすちゅう}) — more compact and noun-bound."
  - slug: "made-ni"
    label: "までに"
    distinction: "までに sets a hard deadline you must beat (五時{ごじ}までに). The 'within-a-period' 〜中{ちゅう}に is softer — sometime inside the period (今日中{きょうじゅう}に = 'at some point today')."
notes:
  - text: "Reading split: 〜ちゅう = 'in the middle of / within' (会議中{かいぎちゅう}, 今日中{きょうじゅう}), but 〜じゅう = 'throughout / all over' (一日中{いちにちじゅう} = 'all day long,' 世界中{せかいじゅう} = 'all over the world'). The kanji is the same; the reading carries the difference."
