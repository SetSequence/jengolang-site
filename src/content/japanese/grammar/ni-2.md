restrictions:

  - text: "Relative time words (今日{きょう}, 明日{あした}, 毎日{まいにち}, 来年{らいねん}) take no に — only absolute times do (七時{しちじ}に ✓, ×明日{あした}に)."

contrasts:

  - slug: "e"
    label: "へ"
